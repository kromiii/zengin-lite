def each_bank(&block)
  Database.each_bank do |row|
    block.call(Bank.from_row(row))
  end
end

def search_banks(name: nil, kana: nil, limit: 10)
  Database.search_banks(name: name, kana: kana, limit: limit).map do |row|
    Bank.from_row(row)
  end
end
